# 简单示例

<template>
  <file-uploader :options="options" />
</template>

<script>
import Vue from 'vue'
import { isEmpty, getTimestamp } from 'npmfe/utils' // 引入工具函数
import { FileUploader } from 'npmfe/components' // 引入组件
import directives from 'npmfe/directives' // 引入指令

directives.forEach((directive) => {
  Vue.use(directive)
})

export default {
  name: 'Example',
  components: {
    FileUploader
  },
  data() {
    return {
      options: {
        crop: true
      }
    }
  },
  mounted() {
    console.log(isEmpty('hello'))
    console.log(getTimestamp(new Date()))
  }
}
</script>
