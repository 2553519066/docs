# StrengthTip 密码强度提示

密码强度验证提示组件。

# 属性

# value

  • 类型：{String}
  • 说明：输入的密码
  • 默认值：''

# 插槽

# default

  • 说明：提示依赖的组件

# 示例

<template>
  <strength-tip :value="password">
    <el-input v-model="password" />
  </strength-tip>
</template>

<script>
  import { StrengthTip } from 'npmfe/components'

  export default {
    name: 'StrengthTipExample',
    components: {
      StrengthTip
    },
    data() {
      return {
        password: ''
      }
    }
  }
</script>
